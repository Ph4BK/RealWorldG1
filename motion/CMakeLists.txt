cmake_minimum_required(VERSION 3.16)
project(g1_motion)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(unitree_api REQUIRED)
add_executable(g1_motion main.cpp)
target_compile_features(g1_motion PRIVATE cxx_std_17)
set(OFFICIAL_INCLUDE /root/unitree_ros2/example/src/include)
target_include_directories(g1_motion PRIVATE ${OFFICIAL_INCLUDE} ${OFFICIAL_INCLUDE}/common ${OFFICIAL_INCLUDE}/nlohmann)
ament_target_dependencies(g1_motion rclcpp nav_msgs geometry_msgs unitree_api)
